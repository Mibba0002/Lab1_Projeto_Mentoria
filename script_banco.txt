CREATE SCHEMA IF NOT EXISTS `mydb` DEFAULT CHARACTER SET utf8 ;
USE `mydb` ;

-- -----------------------------------------------------
-- Table `mydb`.`Mentor`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`Mentor` (
  `cpf_mentor` VARCHAR(11) NOT NULL,
  `nome` VARCHAR(100) NOT NULL,
  `email` VARCHAR(100) NOT NULL,
  `senha` VARCHAR(100) NOT NULL,
  `mini_biografia` TEXT NOT NULL,
  `disponibilidade` VARCHAR(100) NOT NULL,
  `formato_mentoria` VARCHAR(50) NOT NULL,
  `link_portifolio` VARCHAR(200) NOT NULL,
  `redes_profissionais` VARCHAR(200) NOT NULL,
  `status` VARCHAR(30) NOT NULL,
  PRIMARY KEY (`cpf_mentor`),
  UNIQUE INDEX `email_UNIQUE` (`email` ASC) VISIBLE)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`Mentorado`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`Mentorado` (
  `cpf_mentorado` VARCHAR(11) NOT NULL,
  `nome` VARCHAR(100) NOT NULL,
  `email` VARCHAR(100) NOT NULL,
  `senha` VARCHAR(100) NOT NULL,
  `formacao` VARCHAR(100) NOT NULL,
  `objetivos_profissionais` TEXT NOT NULL,
  `principais_duvidas` TEXT NOT NULL,
  `expectativas` TEXT NOT NULL,
  PRIMARY KEY (`cpf_mentorado`),
  UNIQUE INDEX `email_UNIQUE` (`email` ASC) VISIBLE)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`area_atuacao`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`area_atuacao` (
  `id_area_atuacao` INT NOT NULL AUTO_INCREMENT,
  `nome_area` VARCHAR(100) NOT NULL,
  PRIMARY KEY (`id_area_atuacao`))
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`Especializacao`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`Especializacao` (
  `id_area_atuacao` INT NOT NULL,
  `cpf_mentor` VARCHAR(11) NOT NULL,
  `especializacao` VARCHAR(100) NOT NULL,
  `tempo_experiencia` INT NOT NULL,
  PRIMARY KEY (`id_area_atuacao`, `cpf_mentor`),
  INDEX `fk_area_atuacao_has_Mentor_Mentor1_idx` (`cpf_mentor` ASC) VISIBLE,
  INDEX `fk_area_atuacao_has_Mentor_area_atuacao1_idx` (`id_area_atuacao` ASC) VISIBLE,
  CONSTRAINT `fk_area_atuacao_has_Mentor_area_atuacao1`
    FOREIGN KEY (`id_area_atuacao`)
    REFERENCES `mydb`.`area_atuacao` (`id_area_atuacao`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
  CONSTRAINT `fk_area_atuacao_has_Mentor_Mentor1`
    FOREIGN KEY (`cpf_mentor`)
    REFERENCES `mydb`.`Mentor` (`cpf_mentor`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`Mentoria`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`Mentoria` (
  `id_mentoria` INT NOT NULL AUTO_INCREMENT,
  `cpf_mentorado` VARCHAR(11) NOT NULL,
  `id_especializacao` VARCHAR(45) NOT NULL,
  `status` VARCHAR(30) NOT NULL,
  `data_inicio` DATE NOT NULL,
  `data_fim` DATE NOT NULL,
  `objetivos_definidos` TEXT NOT NULL,
  `Depoimentos` TEXT NULL,
  `Especializacao_id_area_atuacao` INT NOT NULL,
  `Especializacao_cpf_mentor` VARCHAR(11) NOT NULL,
  PRIMARY KEY (`id_mentoria`),
  INDEX `fk_Mentoria_Mentorado1_idx` (`cpf_mentorado` ASC) VISIBLE,
  INDEX `fk_Mentoria_Especializacao1_idx` (`Especializacao_id_area_atuacao` ASC, `Especializacao_cpf_mentor` ASC) VISIBLE,
  CONSTRAINT `fk_Mentoria_Mentorado1`
    FOREIGN KEY (`cpf_mentorado`)
    REFERENCES `mydb`.`Mentorado` (`cpf_mentorado`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
  CONSTRAINT `fk_Mentoria_Especializacao1`
    FOREIGN KEY (`Especializacao_id_area_atuacao` , `Especializacao_cpf_mentor`)
    REFERENCES `mydb`.`Especializacao` (`id_area_atuacao` , `cpf_mentor`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`Encontro`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`Encontro` (
  `id_encontro` INT NOT NULL AUTO_INCREMENT,
  `id_mentoria` INT NOT NULL,
  `data` DATE NOT NULL,
  `horario` TIME NOT NULL,
  `tipo_encontro` VARCHAR(20) NOT NULL,
  `descricao` TEXT NOT NULL,
  `link_reuniao` VARCHAR(200) NOT NULL,
  PRIMARY KEY (`id_encontro`),
  INDEX `fk_Encontro_Mentoria1_idx` (`id_mentoria` ASC) VISIBLE,
  CONSTRAINT `fk_Encontro_Mentoria1`
    FOREIGN KEY (`id_mentoria`)
    REFERENCES `mydb`.`Mentoria` (`id_mentoria`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`Interesse`
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS `mydb`.`interesse` (
  `id_area_atuacao` INT NOT NULL,
  `cpf_mentorado` VARCHAR(11) NOT NULL,
  `area_interesse` VARCHAR(100) NOT NULL,
  `nivel_experiencia` VARCHAR(50) NOT NULL,
  PRIMARY KEY (`id_area_atuacao`, `cpf_mentorado`),
  INDEX `fk_area_atuacao_has_Mentorado_Mentorado1_idx` (`cpf_mentorado` ASC) VISIBLE,
  INDEX `fk_area_atuacao_has_Mentorado_area_atuacao1_idx` (`id_area_atuacao` ASC) VISIBLE,
  CONSTRAINT `fk_area_atuacao_has_Mentorado_area_atuacao1`
    FOREIGN KEY (`id_area_atuacao`)
    REFERENCES `mydb`.`area_atuacao` (`id_area_atuacao`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
  CONSTRAINT `fk_area_atuacao_has_Mentorado_Mentorado1`
    FOREIGN KEY (`cpf_mentorado`)
    REFERENCES `mydb`.`Mentorado` (`cpf_mentorado`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB;
